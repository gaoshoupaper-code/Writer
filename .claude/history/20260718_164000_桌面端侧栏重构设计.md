### R1 带来的免费灰度能力

> **executor loader 在本次改造里可以先不改**——API 缺省返回 evolution 的激活配置，而迁移脚本（D16）把老数据复制了一份到 evolution scope，所以 executor 拉到的还是原来那条。改 loader 让它真正去拉 `?scope=executor` 是**可灰度的增强**，不是阻塞项。

### 迁移脚本设计（D16 复制双份）

```python
def _migrate_llm_configs_scope(conn: sqlite3.Connection) -> None:
    """幂等迁移：给 llm_configs 加 scope 列，并把现有数据复制成两份。

    策略（D16）：
    1. 加 scope 列（DEFAULT 'evolution'，现有行自动归 evolution）
    2. 复制现有所有行，副本的 scope 设为 'executor'
    3. 副本的 is_active 保持与原行一致（复制时原本激活的，副本也激活）

    幂等保证：scope 列已存在则直接返回。
    """
    existing = {row[1] for row in conn.execute("PRAGMA table_info(llm_configs)").fetchall()}
    if "scope" in existing:
        return  # 已迁移

    with _lock:
        # 1. 加列（现有行 scope 默认 'evolution'）
        conn.execute("ALTER TABLE llm_configs ADD COLUMN scope TEXT NOT NULL DEFAULT 'evolution'")
        # 2. 复制双份：原行已在 evolution，再复制一份到 executor
        rows = conn.execute(
            "SELECT name, api_key_enc, base_url, model, is_active, created_at, updated_at FROM llm_configs"
        ).fetchall()
        for name, api_key_enc, base_url, model, is_active, created_at, updated_at in rows:
            conn.execute(
                """INSERT INTO llm_configs
                   (name, api_key_enc, base_url, model, is_active, scope, created_at, updated_at)
                   VALUES (?, ?, ?, ?, ?, 'executor', ?, ?)""",
                (name, api_key_enc, base_url, model, is_active, created_at, updated_at),
            )
        # 3. 索引调整
        conn.execute("DROP INDEX IF EXISTS idx_llm_configs_active")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_llm_configs_scope_active ON llm_configs(scope, is_active)")
        conn.commit()
    logger.info("llm_configs 加 scope 列完成，现有数据已复制双份（evolution + executor）。")
```

注册位置：`db.py:392` 后面加一行 `_migrate_llm_configs_scope(conn)`。

    """幂等迁移：给 llm_configs 加 scope 列，并把现有数据复制成两份。

    策略（D16 + T4）：

    if "scope" in existing:

        rows = conn.execute(
            "SELECT name, api_key_enc, base_url, model, is_active, created_at, updated_at FROM llm_configs"
        ).fetchall()
        for name, api_key_enc, base_url, model, is_active, created_at, updated_at in rows:

                """INSERT INTO llm_configs
                   (name, api_key_enc, base_url, model, is_active, scope, created_at, updated_at)

            )
        conn.execute("DROP INDEX IF EXISTS idx_llm_configs_active")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_llm_configs_scope_active ON llm_configs(scope, is_active)")
        conn.commit()
